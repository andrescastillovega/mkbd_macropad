# ZMK Widget Creation Guide

Based on the ZMK codebase and existing patterns, here's how to create a widget from scratch:

## 1. Widget Header File (`widget_name.h`)
```c
#pragma once
#include <lvgl.h>
#include <zephyr/kernel.h>

struct zmk_widget_name {
    sys_snode_t node;    // For widget list management
    lv_obj_t *obj;       // LVGL object pointer
};

int zmk_widget_name_init(struct zmk_widget_name *widget, lv_obj_t *parent);
lv_obj_t *zmk_widget_name_obj(struct zmk_widget_name *widget);
```

## 2. Widget Implementation (`widget_name.c`)
```c
#include <zephyr/kernel.h>
#include <zmk/display.h>
#include <zmk/event_manager.h>
#include <zmk/events/battery_state_changed.h>  // Example event

static sys_slist_t widgets = SYS_SLIST_STATIC_INIT(&widgets);

struct widget_state {
    uint8_t level;    // Example: battery level
    bool charging;    // Example: charging state
};

static void widget_update_cb(struct widget_state state) {
    struct zmk_widget_name *widget;
    SYS_SLIST_FOR_EACH_CONTAINER(&widgets, widget, node) {
        lv_label_set_text_fmt(widget->obj, "Level: %d%%", state.level);
    }
}

static struct widget_state get_state(const zmk_event_t *eh) {
    return (struct widget_state){
        .level = zmk_battery_state_of_charge(),
        .charging = zmk_usb_is_powered()
    };
}

ZMK_DISPLAY_WIDGET_LISTENER(widget_name, struct widget_state,
                            widget_update_cb, get_state)

ZMK_SUBSCRIPTION(widget_name, zmk_battery_state_changed);

int zmk_widget_name_init(struct zmk_widget_name *widget, lv_obj_t *parent) {
    widget->obj = lv_label_create(parent);
    lv_label_set_text(widget->obj, "Init");
    
    sys_slist_append(&widgets, &widget->node);
    widget_name_init();  // Generated by macro
    return 0;
}

lv_obj_t *zmk_widget_name_obj(struct zmk_widget_name *widget) {
    return widget->obj;
}
```

## 3. Configuration Files

**Kconfig:**
```kconfig
config ZMK_WIDGET_NAME
    bool "Widget description"
    depends on ZMK_DISPLAY
    select LV_USE_LABEL
```

**CMakeLists.txt:**
```cmake
target_sources_ifdef(CONFIG_ZMK_WIDGET_NAME app PRIVATE widget_name.c)
```

## 4. Integration in Status Screen
Add to your status screen init function:
```c
static struct zmk_widget_name name_widget;
zmk_widget_name_init(&name_widget, lv_scr_act());
```
